    <script>
        function openProfile(button) {
            const adminId = button.dataset.adminId;

            if (!adminId) {
                console.error("Admin ID not found");
                return;
            }

            window.location.href =
                "admin/admin.html?id=" + encodeURIComponent(adminId);
        }
    </script>

    <script>
        async function loadAdminName() {
            const { data, error } = await supabaseClient
                .from("profiles")
                .select("id, full_name, profile_image")
                .eq("role", "admin")
                .single();

            if (error) {
                console.error("Error loading admin:", error);
                return;
            }

            const avatarElement =
                document.getElementById("adminAvatarImage");

            if (avatarElement && data.profile_image) {
                avatarElement.innerHTML =
                    '<img src="' + data.profile_image + '" alt="Profile">';
            }

            const nameElement =
                document.querySelector(".user-name");

            if (nameElement) {
                nameElement.textContent = data.full_name;
            }

            const adminButton =
                document.querySelector(".user-card");

            if (adminButton) {
                adminButton.dataset.adminId = data.id;
            }
        }

        loadAdminName();
    </script>
